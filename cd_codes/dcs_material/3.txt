
#include <stdio.h>
#include <stdlib.h>
#include <mpi.h>

// =====================================================
// PARTITION 1: MAIN PROGRAM AND MPI INITIALIZATION
// =====================================================

int main(int argc, char *argv[])
{
    int rank, size;
    int N = 10;
    int *array = NULL;
    int local_n;
    int remainder;
    int *sendcounts = NULL;
    int *displs = NULL;
    int *local_array = NULL;
    int local_sum = 0;
    int total_sum = 0;
    int i, start;

    MPI_Init(&argc, &argv);

    MPI_Comm_rank(MPI_COMM_WORLD, &rank);
    MPI_Comm_size(MPI_COMM_WORLD, &size);

    // =================================================
    // PARTITION 2: ARRAY CREATION BY ROOT PROCESS
    // =================================================

    if (rank == 0)
    {
        array = (int *)malloc(N * sizeof(int));

        if (array == NULL)
        {
            fprintf(stderr, "Memory allocation failed.\n");
            MPI_Abort(MPI_COMM_WORLD, 1);
        }

        printf("Enter %d array elements:\n", N);

        for (i = 0; i < N; i++)
        {
            if (scanf("%d", &array[i]) != 1)
            {
                fprintf(stderr, "Invalid input.\n");
                free(array);
                MPI_Abort(MPI_COMM_WORLD, 1);
            }
        }

        printf("\nNumber of elements (N) = %d\n", N);
        printf("Number of processors (n) = %d\n", size);

        printf("Array: ");
        for (i = 0; i < N; i++)
        {
            printf("%d ", array[i]);
        }
        printf("\n\n");
    }

    // =================================================
    // PARTITION 3: DISTRIBUTE N ELEMENTS AMONG PROCESSORS
    // =================================================

    if (rank == 0)
    {
        sendcounts = (int *)malloc(size * sizeof(int));
        displs = (int *)malloc(size * sizeof(int));

        if (sendcounts == NULL || displs == NULL)
        {
            fprintf(stderr, "Memory allocation failed.\n");
            free(array);
            free(sendcounts);
            free(displs);
            MPI_Abort(MPI_COMM_WORLD, 1);
        }

        start = 0;

        for (i = 0; i < size; i++)
        {
            // Distribute remainder elements to first processors
            sendcounts[i] = N / size;

            if (i < N % size)
                sendcounts[i]++;

            displs[i] = start;
            start += sendcounts[i];
        }
    }

    // Send each processor its element count
    MPI_Scatter(
        sendcounts, 1, MPI_INT,
        &local_n, 1, MPI_INT,
        0, MPI_COMM_WORLD
    );

    local_array = (int *)malloc(
        (local_n > 0 ? local_n : 1) * sizeof(int)
    );

    if (local_array == NULL)
    {
        fprintf(stderr, "Memory allocation failed.\n");
        MPI_Abort(MPI_COMM_WORLD, 1);
    }

    // Scatter the array elements among all processors
    MPI_Scatterv(
        array, sendcounts, displs, MPI_INT,
        local_array, local_n, MPI_INT,
        0, MPI_COMM_WORLD
    );

    // =================================================
    // PARTITION 4: CALCULATE INTERMEDIATE SUMS
    // =================================================

    for (i = 0; i < local_n; i++)
    {
        local_sum += local_array[i];
    }

    // Each process displays its assigned data and sum.
    // Output from different processes may appear in any order.
    printf("Processor %d received: ", rank);

    if (local_n == 0)
        printf("(no elements) ");

    for (i = 0; i < local_n; i++)
    {
        printf("%d ", local_array[i]);
    }

    printf("\nProcessor %d intermediate sum = %d\n\n",
           rank, local_sum);

    // =================================================
    // PARTITION 5: REDUCE INTERMEDIATE SUMS
    // =================================================

    MPI_Reduce(
        &local_sum,
        &total_sum,
        1,
        MPI_INT,
        MPI_SUM,
        0,
        MPI_COMM_WORLD
    );

    if (rank == 0)
    {
        printf("====================================\n");
        printf("FINAL SUM OF ARRAY = %d\n", total_sum);
        printf("====================================\n");

        free(array);
        free(sendcounts);
        free(displs);
    }

    free(local_array);

    // =================================================
    // PARTITION 6: TERMINATE MPI
    // =================================================

    MPI_Finalize();

    return 0;
}
